cmake_minimum_required(VERSION 3.20)
project(rgbd_camera VERSION 0.5.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_components REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(rgbd_camera_interfaces 0.3.0 REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
option(RGBD_CAMERA_ENABLE_OAK "Build the OAK/DepthAI provider" ON)
option(RGBD_CAMERA_ENABLE_REALSENSE "Build the D455/librealsense provider" OFF)
if(NOT RGBD_CAMERA_ENABLE_OAK AND NOT RGBD_CAMERA_ENABLE_REALSENSE)
  message(FATAL_ERROR "Enable at least one camera provider")
endif()

add_library(rgbd_camera_core SHARED src/core.cpp src/imu_combiner.cpp src/clock_recovery.cpp)
target_include_directories(rgbd_camera_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(rgbd_camera_core PRIVATE OpenSSL::Crypto)

set(backend_targets)
add_library(rgbd_camera_providers SHARED src/providers.cpp)
target_link_libraries(rgbd_camera_providers PUBLIC rgbd_camera_core)
if(RGBD_CAMERA_ENABLE_OAK)
  find_package(depthai 3.12.1 EXACT CONFIG REQUIRED)
  add_library(rgbd_camera_oak SHARED src/oak_provider.cpp)
  target_link_libraries(rgbd_camera_oak PUBLIC rgbd_camera_core PRIVATE depthai::core Threads::Threads)
  target_link_libraries(rgbd_camera_providers PRIVATE rgbd_camera_oak)
  target_compile_definitions(rgbd_camera_providers PRIVATE RGBD_CAMERA_HAS_OAK)
  list(APPEND backend_targets rgbd_camera_oak)
endif()
if(RGBD_CAMERA_ENABLE_REALSENSE)
  find_package(realsense2 2.56.5 EXACT CONFIG REQUIRED)
  find_package(OpenCV REQUIRED COMPONENTS core imgproc)
  add_library(rgbd_camera_realsense SHARED src/realsense_provider.cpp src/realsense_geometry.cpp)
  target_include_directories(rgbd_camera_realsense PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(rgbd_camera_realsense PUBLIC rgbd_camera_core
    PRIVATE realsense2::realsense2 ${OpenCV_LIBS} Threads::Threads)
  target_link_libraries(rgbd_camera_providers PRIVATE rgbd_camera_realsense)
  target_compile_definitions(rgbd_camera_providers PRIVATE RGBD_CAMERA_HAS_REALSENSE)
  list(APPEND backend_targets rgbd_camera_realsense)
endif()

add_library(rgbd_camera_component SHARED src/camera_node.cpp src/odometry_output.cpp)
target_link_libraries(rgbd_camera_component PRIVATE rgbd_camera_providers)
ament_target_dependencies(rgbd_camera_component PUBLIC rclcpp rclcpp_components
  rcl_interfaces sensor_msgs rgbd_camera_interfaces tf2 tf2_ros)
rclcpp_components_register_node(rgbd_camera_component
  PLUGIN "rgbd_camera::CameraNode" EXECUTABLE camera_node)

add_executable(camera_discover src/discover.cpp)
target_link_libraries(camera_discover PRIVATE rgbd_camera_providers)

install(TARGETS rgbd_camera_core rgbd_camera_providers ${backend_targets} rgbd_camera_component camera_discover
  EXPORT export_rgbd_camera ARCHIVE DESTINATION lib LIBRARY DESTINATION lib
  RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
install(FILES ../tools/fastdds_local_frames.xml DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_core test/test_core.cpp)
  target_link_libraries(test_core PRIVATE rgbd_camera_core)
  add_test(NAME camera_contract COMMAND test_core)
  add_executable(test_clock_recovery test/test_clock_recovery.cpp)
  target_link_libraries(test_clock_recovery PRIVATE rgbd_camera_core)
  add_test(NAME clock_recovery_contract COMMAND test_clock_recovery)
  add_executable(test_recovery_node test/test_recovery_node.cpp)
  target_link_libraries(test_recovery_node PRIVATE rgbd_camera_component rgbd_camera_core)
  ament_target_dependencies(test_recovery_node PUBLIC rclcpp rgbd_camera_interfaces)
  add_test(NAME camera_recovery_continuation COMMAND test_recovery_node)
  set_tests_properties(camera_recovery_continuation PROPERTIES TIMEOUT 15)
  add_executable(test_odometry_core test/test_odometry_core.cpp)
  target_link_libraries(test_odometry_core PRIVATE rgbd_camera_core)
  add_test(NAME odometry_input_contract COMMAND test_odometry_core)
  add_executable(test_odometry_output test/test_odometry_output.cpp)
  target_link_libraries(test_odometry_output PRIVATE rgbd_camera_component rgbd_camera_core)
  ament_target_dependencies(test_odometry_output PUBLIC rclcpp rgbd_camera_interfaces tf2_ros)
  add_test(NAME odometry_output_contract COMMAND test_odometry_output)
  set_tests_properties(odometry_output_contract PROPERTIES TIMEOUT 10)
  add_executable(test_node test/test_node.cpp)
  target_link_libraries(test_node PRIVATE rgbd_camera_component rgbd_camera_core)
  ament_target_dependencies(test_node PUBLIC rclcpp rgbd_camera_interfaces)
  add_test(NAME camera_adapter COMMAND test_node)
  set_tests_properties(camera_adapter PROPERTIES TIMEOUT 20)
  if(RGBD_CAMERA_ENABLE_REALSENSE)
    add_executable(test_realsense_geometry test/test_realsense_geometry.cpp)
    target_link_libraries(test_realsense_geometry PRIVATE rgbd_camera_realsense realsense2::realsense2)
    add_test(NAME realsense_geometry COMMAND test_realsense_geometry)
  endif()
endif()
ament_export_targets(export_rgbd_camera HAS_LIBRARY_TARGET)
ament_export_dependencies(rclcpp sensor_msgs rgbd_camera_interfaces)
ament_package()
